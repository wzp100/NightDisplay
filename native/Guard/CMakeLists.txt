cmake_minimum_required(VERSION 3.20)
project(NightDisplayGuard LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "The minimal guard currently requires MSVC on Windows x64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The guard requires a Windows x64 target.")
endif()
# Keep Debug builds free of CRT-dependent runtime checks as well.
string(REGEX REPLACE "/RTC[1csu]+" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
add_executable(NightDisplay.Guard WIN32 main.cpp)
target_compile_features(NightDisplay.Guard PRIVATE cxx_std_14)
target_compile_definitions(NightDisplay.Guard PRIVATE UNICODE _UNICODE)
target_compile_options(NightDisplay.Guard PRIVATE /W4 /WX /utf-8 /O1 /Oi /GS- /Zl)
target_link_options(NightDisplay.Guard PRIVATE /ENTRY:GuardMain /NODEFAULTLIB)
target_link_libraries(NightDisplay.Guard PRIVATE kernel32)
